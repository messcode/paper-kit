% ----------------------------------------------------------------------
% Supplementary Note command
%   \suppnote{Title}{Body}
% Produces an unnumbered section "Supplementary Note N: Title" and adds it
% to the Notes sub-list of the Table of Contents.
% ----------------------------------------------------------------------
\newcounter{suppnotenumber}
\newcommand{\suppnote}[2]{%
  \refstepcounter{suppnotenumber}%
  \section*{Supplementary Note \thesuppnotenumber: #1}%
  \addcontentsline{lon}{suppnote}{\protect\numberline{\thesuppnotenumber}#1}%
  #2%
}
